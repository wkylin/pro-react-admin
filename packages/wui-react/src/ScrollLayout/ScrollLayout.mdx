```tsx
import ScrollLayout from '@w.ui/wui-react/scroll-layout'
import '@w.ui/wui-react/style.css'

export function Example() {
  return (
    <div style={{ height: 420 }}>
      <ScrollLayout header={<h2>列表</h2>} footer={<button type="button">保存</button>}>
        <div style={{ height: 900, padding: 24 }}>可滚动内容</div>
      </ScrollLayout>
    </div>
  )
}
```
