<p>
  <code>onAction</code> 收到的动作类型包括 <code>input</code>、<code>paste</code>、<code>backspace</code>、
  <code>next</code> 与 <code>close</code>。输入和粘贴动作带有 <code>text</code>
  。常见做法是把字符插入当前选区；示例为了简洁追加到受控值末尾。
</p>

<pre>
  <code>{`import { VirtualKeyboard, type VirtualKeyboardAction } from '@w.ui/wui-react'

const handleAction = (action: VirtualKeyboardAction) => {
  if (action.type === 'input' || action.type === 'paste') {
    setValue((current) => current + action.text)
  } else if (action.type === 'backspace') {
    setValue((current) => Array.from(current).slice(0, -1).join(''))
  } else if (action.type === 'next') {
    nextFieldRef.current?.focus()
  } else {
    setKeyboardVisible(false)
  }
}

<VirtualKeyboard
  layout="phone"
  visible={keyboardVisible}
  onAction={handleAction}
  onPasteError={() => setMessage('无法读取剪贴板')}
/>`}</code>
</pre>
