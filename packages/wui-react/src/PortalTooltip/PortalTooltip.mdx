### 选择触发方式

<table className="wui-doc-table">
  <thead>
    <tr>
      <th>方式</th>
      <th>属性</th>
      <th>使用建议</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>悬停或焦点</td>
      <td>
        <code>trigger=&quot;hover&quot;</code>
      </td>
      <td>对已有操作给出补充说明</td>
    </tr>
    <tr>
      <td>点击</td>
      <td>
        <code>trigger=&quot;click&quot;</code>
      </td>
      <td>内容需要主动打开或停留更久</td>
    </tr>
    <tr>
      <td>外部控制</td>
      <td>
        <code>visible</code>
      </td>
      <td>
        与页面状态同步；<code>defaultVisible</code> 仅设置初始状态
      </td>
    </tr>
  </tbody>
</table>

```tsx
import PortalTooltip from '@w.ui/wui-react/portal-tooltip'
import '@w.ui/wui-react/style.css'

export function Example() {
  return (
    <PortalTooltip title="查看详细说明" placement="top" trigger="hover">
      <button type="button" aria-label="查看详细说明">
        ?
      </button>
    </PortalTooltip>
  )
}
```
