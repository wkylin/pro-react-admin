### 难度与事件

<table className="wui-doc-table">
  <thead>
    <tr>
      <th>属性</th>
      <th>作用</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>
        <code>difficulty</code>
      </td>
      <td>
        <code>easy</code>、<code>medium</code> 或 <code>hard</code>，默认 <code>medium</code>
      </td>
    </tr>
    <tr>
      <td>
        <code>onSuccess</code>
      </td>
      <td>拼图在客户端判定成功时触发</td>
    </tr>
    <tr>
      <td>
        <code>onFail</code>
      </td>
      <td>拼图在客户端判定失败时触发</td>
    </tr>
    <tr>
      <td>
        <code>onRefresh</code>
      </td>
      <td>生成新拼图后触发</td>
    </tr>
  </tbody>
</table>

import SliderCaptcha from '@w.ui/wui-react/slider-captcha'
