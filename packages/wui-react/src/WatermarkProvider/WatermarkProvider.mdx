### 快速接入

```tsx
import WatermarkProvider from '@w.ui/wui-react/watermark-provider'

export function Example() {
  return (
    <div id="report-panel">
      <WatermarkProvider content="仅供内部使用" containerId="report-panel" opacity={0.08}>
        <div>报表内容</div>
      </WatermarkProvider>
    </div>
  )
}
```
