### 快速接入

```tsx
import { useState } from 'react'
import OneTimePasscode from '@w.ui/wui-react/one-time-passcode'
import '@w.ui/wui-react/style.css'

function VerificationCode() {
  const [code, setCode] = useState('')
  return (
    <OneTimePasscode
      length={6}
      value={code}
      onChange={setCode}
      validateChar={(ch) => ch === '' || /^[0-9]$/.test(ch)}
    />
  )
}
```
