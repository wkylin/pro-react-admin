### 快速接入

```tsx
import OverflowText from '@w.ui/wui-react/overflow-text'
import '@w.ui/wui-react/style.css'

export function Example() {
  return <OverflowText text="一段可能超出容器的标题" maxWidth={240} lines={2} />
}
```
