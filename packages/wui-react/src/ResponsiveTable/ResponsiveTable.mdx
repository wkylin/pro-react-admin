### 按数据来源接入

<table className="wui-doc-table">
  <thead>
    <tr>
      <th>数据来源</th>
      <th>关键属性</th>
      <th>何时使用</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>本地数组</td>
      <td>
        <code>dataSource</code>、<code>columns</code>
      </td>
      <td>数据已在页面中，或数据量较小</td>
    </tr>
    <tr>
      <td>自定义函数</td>
      <td>
        <code>fetchData</code>、<code>autoLoad</code>
      </td>
      <td>业务已有请求函数，需要远程分页</td>
    </tr>
    <tr>
      <td>请求地址</td>
      <td>
        <code>fetchUrl</code>、<code>requestLib</code>
      </td>
      <td>通过指定客户端请求，并控制鉴权与拦截器</td>
    </tr>
  </tbody>
</table>

```tsx
import ResponsiveTable from '@w.ui/wui-react/responsive-table'
import '@w.ui/wui-react/style.css'

export function Example() {
  return (
    <ResponsiveTable
      rowKey="id"
      columns={[{ title: '名称', dataIndex: 'name' }]}
      dataSource={[{ id: 1, name: '示例记录' }]}
    />
  )
}
```
