### 快速接入

```tsx
import SafeHtml from '@w.ui/wui-react/safe-html'

export function Example() {
  const articleHtml = '<p><strong>重点</strong>内容</p>'
  return <SafeHtml html={articleHtml} fallback={<p>暂无内容</p>} />
}
```
